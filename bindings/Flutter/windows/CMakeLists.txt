cmake_minimum_required(VERSION 3.16)
project(rhwp_native_reference LANGUAGES CXX)
option(RHWP_D3D_LIFETIME_TESTS_ONLY "Portable Windows texture lifetime checks without a Windows SDK" OFF)
if(RHWP_D3D_LIFETIME_TESTS_ONLY)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(frame_lifetime_test tests/frame_lifetime_test.cpp)
  target_include_directories(frame_lifetime_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_features(frame_lifetime_test PRIVATE cxx_std_17)
  target_link_libraries(frame_lifetime_test PRIVATE Threads::Threads)
  add_test(NAME windows_frame_lifetime COMMAND frame_lifetime_test)
  add_executable(import_lifetime_test tests/import_lifetime_test.cpp)
  target_include_directories(import_lifetime_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_features(import_lifetime_test PRIVATE cxx_std_17)
  target_link_libraries(import_lifetime_test PRIVATE Threads::Threads)
  add_test(NAME windows_import_lifetime COMMAND import_lifetime_test)
  add_executable(close_registry_test tests/close_registry_test.cpp)
  target_include_directories(close_registry_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_features(close_registry_test PRIVATE cxx_std_17)
  add_test(NAME windows_close_registry COMMAND close_registry_test)
  return()
endif()
if(NOT WIN32)
  message(FATAL_ERROR "The D3D plugin requires Windows; use RHWP_D3D_LIFETIME_TESTS_ONLY for portable checks")
endif()
add_library(rhwp_native_reference_plugin SHARED rhwp_d3d_plugin.cpp Producer.cpp NativeApi.cpp)
get_filename_component(RHWP_PLUGIN_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
if(COMMAND apply_standard_settings)
  apply_standard_settings(rhwp_native_reference_plugin)
endif()
target_compile_features(rhwp_native_reference_plugin PRIVATE cxx_std_17)
target_compile_definitions(rhwp_native_reference_plugin PRIVATE RHWP_NATIVE_REFERENCE_PLUGIN_IMPL NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(rhwp_native_reference_plugin PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(rhwp_native_reference_plugin PRIVATE "${RHWP_PLUGIN_SOURCE_DIR}/../../swift/Sources/CRhwpNative")
target_link_libraries(rhwp_native_reference_plugin PRIVATE flutter flutter_wrapper_plugin d3d12 d3d11 dxgi dxguid)
set(rhwp_native_reference_bundled_libraries "" PARENT_SCOPE)
