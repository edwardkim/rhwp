use std::{env, fs, path::PathBuf};
use rhwp::document_core::DocumentCore;
use rhwp::paint::RenderProfile;
use rhwp::renderer::{font_environment::FontEnvironment, skia::SkiaLayerRenderer};
use rhwp::renderer::layer_renderer::{LayerRasterRenderer, RasterRenderOptions};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let args: Vec<String> = env::args().collect();
    let mut core = DocumentCore::from_bytes(&fs::read(&args[1])?)?;
    core.set_hangul2024_compat(false);
    core.set_font_environment(Some(FontEnvironment::from_json(&fs::read_to_string(&args[2])?)?))?;
    let paths: Vec<PathBuf> = args[4..].iter().map(PathBuf::from).collect();
    let renderer = SkiaLayerRenderer::new().with_font_paths(&paths);
    renderer.prepare_declared_font_fallbacks(&mut core)?;
    renderer.prepare_native_pua_fonts(&mut core)?;
    renderer.prepare_native_otf_fonts(&mut core)?;
    renderer.prepare_text_shadow_fonts(&mut core)?;
    renderer.prepare_metafile_fonts(&mut core)?;
    fs::create_dir_all(&args[3])?;
    for page in [1u32, 4, 11] {
        core.set_hangul2024_compat(false);
        let mut tree = core.build_page_layer_tree_with_profile(page, RenderProfile::Print)?;
        let original = tree.to_json();
        assert!(!tree.preserves_raster_text_geometry());
        core.set_hangul2024_compat(true);
        let geometry = core.build_page_layer_tree_with_profile(page, RenderProfile::Print)?;
        // 조판과 원점은 그대로 두고 자원의 글자 재생 정책만 분리한다.
        tree.resources = geometry.resources;
        assert!(tree.preserves_raster_text_geometry());
        assert_eq!(original.replace("\"rasterTextGeometry\":false", "\"rasterTextGeometry\":true"), tree.to_json());
        let png = renderer.render_png_with_options(&tree, RasterRenderOptions::default())?;
        let out = PathBuf::from(&args[3]).join(format!("geometry_{:03}.png", page + 1));
        fs::write(&out, png)?;
        println!("동일 조판·자원, 글자 재생 정책만 변경: {}", out.display());
    }
    Ok(())
}
