const out=root+'/output/pr-review/pr9063';

 ['/rhwp.js',['application/javascript',readFileSync(pkg+'/rhwp.js')]],

});

let browser;
try{
 browser=await require('puppeteer-core').launch({executablePath:findChrome(process.env.VISUAL_SWEEP_CHROME),headless:true,protocolTimeout:180000});
 const page=await browser.newPage();

 const results=[];
 for(const name of ['picture_in_front_over_1px','rect_top_and_bottom_over_12px','picture_in_front_fits']){

   const doc=new rhwp.HwpDocument(new Uint8Array(bytes));doc.setHangul2024Compat(false);doc.selectCanvasMetrics(false);doc.setFontEnvironment(globalThis.probeFontEnvironment);
   function snapshot(label){
    const count=doc.pageCount(),pages=[];
    for(let i=0;i<count;i++){
     const tree=JSON.parse(doc.getPageRenderTree(i));const objects=[];
     function walk(n){if(['Image','Rect','Rectangle'].includes(n.type))objects.push({kind:n.type,...n.bbox});for(const c of n.children||[])walk(c);}
     walk(tree.root||tree);pages.push({page:i+1,objects});
    }
    return {label,page_count:count,pages};
   }
   const states=[snapshot('loaded')];doc.insertText(0,0,0,'X');states.push(snapshot('edited'));
   const saved=doc.exportHwp();doc.free();
   const reopened=new rhwp.HwpDocument(saved);reopened.setFontEnvironment(globalThis.probeFontEnvironment);const reopenedCount=reopened.pageCount();reopened.free();
   return {states,reopened_page_count:reopenedCount};
  },Array.from(bytes));
  results.push({input:name+'.hwp',...result});
  console.log(name,JSON.stringify(result));
 }
