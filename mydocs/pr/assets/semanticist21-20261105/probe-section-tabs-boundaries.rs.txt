
use rhwp::document_core::DocumentCore;
fn x(core: &DocumentCore, section: usize, offset: usize) -> f64 {
    let text=core.get_cursor_rect_native(section, 1, offset).unwrap();
    let rest=text.split("\"x\":").nth(1).unwrap();
    rest.split([',','}']).next().unwrap().parse().unwrap()
}
fn gap(core: &DocumentCore, section: usize, offset: usize) -> f64 {x(core,section,offset)-x(core,section,0)}
fn near(a:f64,b:f64,label:&str){assert!((a-b).abs()<0.11,"{label}: {a} != {b}");}
fn main(){
 let args:Vec<_>=std::env::args().collect();let bytes=std::fs::read(&args[1]).unwrap();
 let mut one=DocumentCore::from_bytes(&bytes).unwrap();
 let mut small=DocumentCore::from_bytes(&bytes).unwrap();small.set_section_def_native(0,r#"{"defaultTabSpacing":8000}"#).unwrap();
 let mut multi=DocumentCore::from_bytes(&bytes).unwrap();let mut doc=multi.document().clone();doc.sections.push(doc.sections[0].clone());multi.set_document(doc);multi.set_section_def_native(1,r#"{"defaultTabSpacing":8000}"#).unwrap();
 for order in [[0,1],[1,0],[0,1]] {for sec in order {let expected=if sec==0 {gap(&one,0,2)}else{gap(&small,0,2)};let got=gap(&multi,sec,2);near(got,expected,"cold/warm alternating sections");println!("section={sec} actual={got} standalone={expected}");}}
 multi.insert_text_native(1,1,0,"A").unwrap();small.insert_text_native(0,1,0,"A").unwrap();near(gap(&multi,1,3),gap(&small,0,3),"incremental paragraph edit");
 multi.set_section_def_native(0,r#"{"defaultTabSpacing":24000}"#).unwrap();one.set_section_def_native(0,r#"{"defaultTabSpacing":24000}"#).unwrap();near(gap(&multi,0,2),gap(&one,0,2),"section zero edit");near(gap(&multi,1,3),gap(&small,0,3),"unrelated section preserved");
 let saved=rhwp::serializer::hwpx::serialize_hwpx(multi.document()).unwrap();let reopened=DocumentCore::from_bytes(&saved).unwrap();near(gap(&reopened,0,2),gap(&one,0,2),"HWPX reopen section zero");near(gap(&reopened,1,3),gap(&small,0,3),"HWPX reopen section one");
 println!("PASS: independent section controls, alternating cold/warm, incremental edit, section edit, HWPX reopen");
}
