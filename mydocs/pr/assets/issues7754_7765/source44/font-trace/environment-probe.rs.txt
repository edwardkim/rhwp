use rhwp::{DocumentCore,renderer::font_environment::FontEnvironment};
use rhwp::document_core::queries::{png_export::NativePngExporter,rendering::PngExportOptions};
use rhwp::paint::RenderProfile;
use std::{fs,path::PathBuf};
fn main(){let a:Vec<String>=std::env::args().collect();let mut doc=DocumentCore::from_bytes(&fs::read(&a[1]).unwrap()).unwrap();let page:u32=a[2].parse().unwrap();let mut paths:Vec<PathBuf>=a[5..].iter().map(PathBuf::from).collect();let env=if a[3]=="auto"{let supply=rhwp::renderer::native_webfonts::prepare(&doc.document().doc_info,&paths).unwrap();paths.extend(supply.paths.iter().cloned());let native=FontEnvironment::from_native_document_substitutions(&doc.document().doc_info,&paths).unwrap();supply.merge_into_environment(native).unwrap()}else{Some(FontEnvironment::from_json(&fs::read_to_string(&a[3]).unwrap()).unwrap())};fs::write(format!("{}.environment.json",a[4]),serde_json::to_vec_pretty(&env).unwrap()).unwrap();doc.set_font_environment(env).unwrap();let exp=NativePngExporter::new(RenderProfile::Print,PngExportOptions{dpi:Some(96.0),font_paths:paths,..Default::default()}).with_font_fallback_reporting(true);exp.prepare_document(&mut doc).unwrap();let png=exp.render_page(&doc,page-1).unwrap();fs::write(format!("{}.png",a[4]),png).unwrap();let trace=doc.get_font_decision_trace_native(page-1,r#"{"maxCharacters":4096}"#).unwrap();fs::write(format!("{}.trace.json",a[4]),trace).unwrap();fs::write(format!("{}.paint.json",a[4]),serde_json::to_vec_pretty(&exp.font_fallback_report()).unwrap()).unwrap();}
